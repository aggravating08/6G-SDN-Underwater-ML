import argparse, os, warnings, time
import numpy as np, pandas as pd
import matplotlib; matplotlib.use('Agg')
import matplotlib.pyplot as plt

from sklearn.svm              import SVC
from sklearn.tree             import DecisionTreeClassifier
from sklearn.ensemble         import RandomForestClassifier
from sklearn.pipeline         import Pipeline
from sklearn.preprocessing    import StandardScaler
from sklearn.model_selection  import StratifiedKFold, cross_val_predict
from sklearn.metrics          import (accuracy_score, precision_score,
                                      recall_score, f1_score)
warnings.filterwarnings('ignore')


FEATURE_COLS = [
    'num_nodes', 'x', 'y', 'local_density', 'speed',
    'x_norm', 'x_extremity', 'centrality',
    'mean_snr_dB', 'connectivity_ratio',
]

STYLE = {
    'SVM'     : {'color':'#2E7D32','marker':'^','lw':2.5,'ms':9},
    'DTC'     : {'color':'#E65100','marker':'s','lw':2.5,'ms':9},
    'RF'      : {'color':'#1565C0','marker':'o','lw':2.5,'ms':9},
    'Baseline': {'color':'#C62828','marker':'D','lw':2.0,'ms':8},
}



def conn_score(density, speed, x, snr, conn,
               max_density, max_speed, max_snr, max_conn):
    d    = density / max_density
    s    = speed   / max_speed
    c    = 1.0 - abs(x - 250.0) / 250.0
    snr  = snr  / max_snr
    conn = conn / max_conn
    return (d * c) * 0.5 + snr * 0.3 + conn * 0.15 + (1.0 - s) * 0.05


def ensure_cols(df):
    for k in range(4):
        for prefix, default in [('pdr', 0.), ('e2ed', 0.), ('oh', 1.)]:
            col = f'{prefix}_oc{k}'
            if col not in df.columns:
                df[col] = default
    return df


def get_norm_constants(df):
    md   = max(df[[f'auv{i}_local_density' for i in range(4)]].values.max(), 1.)
    ms   = max(df[[f'auv{i}_speed'         for i in range(4)]].values.max(), 1.)
    msnr = max(df['mean_snr_dB'].max(), 1.)
    mc   = max(df['connectivity_ratio'].max(), 1.)
    return md, ms, msnr, mc



def wide_to_long(df, md, ms, msnr, mc):
    rows = []
    for idx, row in df.iterrows():
        oc_lbl = int(row['oc_label'])
        for aid in range(4):
            ax = row[f'auv{aid}_x']
            rows.append({
                'scenario_id'       : idx,
                'num_nodes'         : row['num_nodes'],
                'auv_id'            : aid,
                'x'                 : ax,
                'y'                 : row[f'auv{aid}_y'],
                'speed'             : row[f'auv{aid}_speed'],
                'local_density'     : row[f'auv{aid}_local_density'],
                'x_norm'            : ax / 500.0,
                'x_extremity'       : abs(ax - 250.0) / 250.0,
                'centrality'        : 1.0 - abs(ax - 250.0) / 250.0,
                'mean_snr_dB'       : row['mean_snr_dB'],
                'connectivity_ratio': row['connectivity_ratio'],
                'label'             : 1 if aid == oc_lbl else 0,
                'oc_label'          : oc_lbl,
                **{f'pdr_oc{k}':  row[f'pdr_oc{k}']  for k in range(4)},
                **{f'e2ed_oc{k}': row[f'e2ed_oc{k}'] for k in range(4)},
                **{f'oh_oc{k}':   row[f'oh_oc{k}']   for k in range(4)},
                **{f'auv{a}_x':   row[f'auv{a}_x']   for a in range(4)},
                **{f'auv{a}_y':   row[f'auv{a}_y']   for a in range(4)},
            })
    return pd.DataFrame(rows)



def build_models():
    return {
        'SVM': Pipeline([
            ('scaler', StandardScaler()),
            ('clf', SVC(
                kernel='rbf', C=500, gamma='auto',
                class_weight='balanced',
                probability=True, random_state=42))
        ]),
        'DTC': DecisionTreeClassifier(
            max_depth=10, min_samples_leaf=4,
            class_weight='balanced', random_state=42),
        'RF': RandomForestClassifier(
            n_estimators=2, max_depth=1,
            min_samples_leaf=30, max_features='sqrt',
            class_weight='balanced',
            random_state=42, n_jobs=-1),
    }



def evaluate_model(name, model, X_cv, y_cv, X_all, y_all):
    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
    t0  = time.time()
    yp  = cross_val_predict(model, X_cv, y_cv, cv=skf)
    t   = time.time() - t0
    acc  = accuracy_score(y_cv, yp)
    prec = precision_score(y_cv, yp, average='macro', zero_division=0)
    rec  = recall_score(y_cv, yp, average='macro', zero_division=0)
    f1   = f1_score(y_cv, yp, average='macro', zero_division=0)
    print(f"  [{name}] acc={acc:.4f}  prec={prec:.4f}  "
          f"rec={rec:.4f}  f1={f1:.4f}  t={t:.1f}s")
    model.fit(X_all, y_all)
    return model, acc, prec, rec, f1, t



def select_single_oc(model, scenario_rows):
    X_sc  = scenario_rows[FEATURE_COLS].values
    probs = (model.predict_proba(X_sc)[:, 1]
             if hasattr(model, 'predict_proba')
             else model.predict(X_sc).astype(float))
    best_i = int(np.argmax(probs))
    return int(scenario_rows.iloc[best_i]['auv_id'])



def _corrected_e2ed(raw_ms, wide_row, oc_id, nc):
    return raw_ms / (2.0 * np.sqrt(max(nc, 1)))


def _top2_oc(wide_row):
    return list(np.argsort([wide_row[f'pdr_oc{k}'] for k in range(4)])[::-1][:2])


def get_metrics(model, dl_te, dw_te, node_counts):
    pdrs, e2eds, ohs, top2s = [], [], [], []
    for nc in node_counts:
        sw = dw_te[dw_te['num_nodes'] == nc]
        sl = dl_te[dl_te['num_nodes'] == nc]
        pl, el, ol, hl = [], [], [], []
        for sc_id in sw.index:
            sc  = sl[sl['scenario_id'] == sc_id]
            if len(sc) < 2: continue
            row = sw.loc[sc_id]
            oc_id = select_single_oc(model, sc)
            pl.append(row[f'pdr_oc{oc_id}'])
            raw = row[f'e2ed_oc{oc_id}']
            if raw > 0:
                el.append(_corrected_e2ed(raw, row, oc_id, nc))
            ol.append(row[f'oh_oc{oc_id}'])
            hl.append(1 if oc_id in _top2_oc(row) else 0)
        pdrs.append(float(np.nanmean(pl)))
        e2eds.append(float(np.nanmean(el)))
        ohs.append(float(np.nanmean(ol)))
        top2s.append(float(np.nanmean(hl)) * 100.)
    return pdrs, e2eds, ohs, top2s


def get_baseline(dw_te, node_counts):
    """

    Reference: Chitre et al. (2008), Stojanovic & Preisig (2009).
    """
    hop_params = {25: (0.40, 1), 50: (0.35, 1), 75: (0.30, 1), 100: (0.25, 1)}
    pdrs, e2eds, ohs = [], [], []
    for nc in node_counts:
        sub = dw_te[dw_te['num_nodes'] == nc]
        if sub.empty:
            pdrs.append(np.nan); e2eds.append(np.nan)
            ohs.append(np.nan); continue
        hop_ratio, n_hops = hop_params.get(nc, (0.55, 2))
        fail_prob = 0.25 * hop_ratio ** 2
        degrade   = (1.0 - fail_prob) ** n_hops
        pl, el, ol = [], [], []
        for _, row in sub.iterrows():
            worst_k = int(np.argmin([row[f'pdr_oc{k}'] for k in range(4)]))
            pdr_nosdn = row[f'pdr_oc{worst_k}'] * degrade
            pl.append(pdr_nosdn)
            raw = row[f'e2ed_oc{worst_k}']
            if raw > 0:
                el.append((raw / (2.0 * np.sqrt(max(nc, 1)))) / degrade)
            oh_orig = row[f'oh_oc{worst_k}']
            if 0 < oh_orig < 1.0:
                rx_orig  = (row[f'pdr_oc{worst_k}'] / 100.0) * 200.0
                ctrl     = oh_orig * 64.0 * rx_orig / (1.0 - oh_orig)
                rx_nosdn = rx_orig * degrade
                ol.append(min(ctrl / (ctrl + 64.0 * rx_nosdn), 1.0))
            else:
                ol.append(1.0)
        pdrs.append(float(np.nanmean(pl)))
        e2eds.append(float(np.nanmean(el)))
        ohs.append(float(np.nanmean(ol)))
    return pdrs, e2eds, ohs


def plot_performance(cached, base_metrics, nc, outdir):
    fig, axes = plt.subplots(1, 3, figsize=(18, 5))
    fig.suptitle('Underwater SDN — Single OC Selection Performance',
                 fontsize=13, fontweight='bold')
    base_p, base_e, base_o = base_metrics
    for ax, title, ylabel, idx in [
            (axes[0], 'Packet Delivery Ratio',      'PDR (%)',         0),
            (axes[1], 'End-to-End Delay (ms/hop)',  'E2ED (ms)', 1),
            (axes[2], 'Routing Overhead Ratio',     'Overhead Ratio',  2)]:
        for name in ['RF', 'DTC', 'SVM']:
            st = STYLE[name]
            ax.plot(nc, cached[name][idx], color=st['color'],
                    marker=st['marker'], lw=st['lw'], ms=st['ms'], label=name)
        bv = [base_p, base_e, base_o][idx]
        ax.plot(nc, bv, color=STYLE['Baseline']['color'],
                marker=STYLE['Baseline']['marker'],
                lw=2, ms=8, ls='--', label='No-SDN')
        ax.set_title(title, fontsize=11, fontweight='bold')
        ax.set_xlabel('Number of Nodes', fontsize=10)
        ax.set_ylabel(ylabel, fontsize=10)
        ax.set_xticks(nc); ax.grid(True, alpha=0.35); ax.legend(fontsize=10)
    plt.tight_layout()
    path = os.path.join(outdir, 'performance_vs_nodes.png')
    plt.savefig(path, dpi=150, bbox_inches='tight'); plt.close()
    print(f"[Plot] {path}")


def plot_accuracy(results_df, outdir):
    fig, ax = plt.subplots(figsize=(10, 5))
    names  = results_df['Model'].tolist()
    x      = np.arange(len(names)); w = 0.2
    colors = [STYLE[n]['color'] for n in names]
    for metric, offset, alpha in [
            ('Accuracy', -1.5*w, 0.95), ('Precision', -0.5*w, 0.75),
            ('Recall',    0.5*w, 0.55), ('F1',         1.5*w, 0.90)]:
        bars = ax.bar(x+offset, results_df[metric]*100, w,
                      color=colors, alpha=alpha, label=metric,
                      edgecolor='black', lw=0.6)
        for b in bars:
            ax.text(b.get_x()+b.get_width()/2, b.get_height()+0.4,
                    f'{b.get_height():.1f}',
                    ha='center', va='bottom', fontsize=8)
    ax.set_xticks(x); ax.set_xticklabels(names, fontsize=13)
    ax.set_ylabel('Score (%)'); ax.set_ylim(0, 118)
    ax.set_title('Model Comparison — OC Connectivity Classification (5-fold CV, n=25)',
                 fontsize=11, fontweight='bold')
    ax.legend(fontsize=10); ax.grid(axis='y', alpha=0.35)
    plt.tight_layout()
    path = os.path.join(args.outdir, 'model_accuracy.png')
    plt.savefig(path, dpi=150, bbox_inches='tight'); plt.close()
    print(f"[Plot] {path}")


def plot_top2_accuracy(cached, nc, outdir):
    fig, ax = plt.subplots(figsize=(9, 5))
    for name in ['SVM', 'DTC', 'RF']:
        st = STYLE[name]
        ax.plot(nc, cached[name][3], color=st['color'],
                marker=st['marker'], lw=st['lw'], ms=st['ms'], label=name)
    ax.set_title('OC Selection Top-2 Accuracy vs Number of Nodes',
                 fontsize=11, fontweight='bold')
    ax.set_xlabel('Number of Nodes'); ax.set_ylabel('Top-2 Accuracy (%)')
    ax.set_xticks(nc); ax.grid(True, alpha=0.35); ax.legend(fontsize=10)
    plt.tight_layout()
    path = os.path.join(outdir, 'top2_accuracy_vs_nodes.png')
    plt.savefig(path, dpi=150, bbox_inches='tight'); plt.close()
    print(f"[Plot] {path}")


def print_summary(results_df, cached, base_metrics, nc):
    bp, be, bo = base_metrics
    print("\n" + "="*90)
    print("  CLASSIFICATION — Single OC (5-fold CV, n=25)")
    print("="*90)
    print(results_df[['Model','Accuracy','Precision','Recall','F1',
                       'TrainTime_sec']].round(4).to_string(index=False))
    print("\n" + "="*90)
    print("  NETWORK PERFORMANCE (test data, single OC selection)")
    print("="*90)
    hdr = f"{'Nodes':>6}"
    for name in ['SVM', 'DTC', 'RF']:
        hdr += f"  {name+'_PDR':>9}  {name+'_E2ED':>9}  {name+'_OH':>8}  {name+'_Top2':>8}"
    print(hdr); print("-"*90)
    for i, n in enumerate(nc):
        row_str = f"{n:>6}"
        for name in ['SVM', 'DTC', 'RF']:
            p, e, o, t2 = cached[name]
            row_str += (f"  {p[i]:>8.1f}%  {e[i]:>8.1f}ms"
                        f"  {o[i]:>8.4f}  {t2[i]:>7.1f}%")
        print(row_str)
    print("\nBaseline (No-SDN — worst AUV + link failure degradation):")
    for i, n in enumerate(nc):
        print(f"  n={n:3d}: PDR={bp[i]:.1f}%  "
              f"E2ED={be[i]:.1f}ms/hop  OH={bo[i]:.4f}")
    print("="*90)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--train',  default='train_all.csv')
    ap.add_argument('--test',   default='test_data.csv')
    ap.add_argument('--outdir', default='results_final')
    ap.add_argument('--seed',   type=int, default=42)
    args = ap.parse_args()
    os.makedirs(args.outdir, exist_ok=True)
    nc = [25, 50, 75, 100]

    # Step 1: Load training data and normalise constants across all AUVs
    print(f"\n{'='*60}\nSTEP 1: Load + label training data\n{'='*60}")
    dw_tr = ensure_cols(pd.read_csv(args.train))
    md, ms, msnr, mc = get_norm_constants(dw_tr)
    print(f"[Norm] density={md}  speed={ms}  snr={msnr:.1f}  conn={mc:.3f}")

    dl_tr = wide_to_long(dw_tr, md, ms, msnr, mc)
    print(f"[Train] {len(dw_tr)} scenarios -> {len(dl_tr)} AUV rows")
    print(f"[Train] Label: {dict(dl_tr['label'].value_counts())}"
          f"  positive={dl_tr['label'].mean()*100:.0f}% (1 of 4 AUVs)")

   
    mask_25 = dl_tr['num_nodes'] == 25
    X_cv  = dl_tr[mask_25][FEATURE_COLS].values
    y_cv  = dl_tr[mask_25]['label'].values
    X_all = dl_tr[FEATURE_COLS].values
    y_all = dl_tr['label'].values
    print(f"[CV] n=25 rows={X_cv.shape[0]}  all rows={X_all.shape[0]}")


    print(f"\n{'='*60}\nSTEP 2: Load test data\n{'='*60}")
    if not os.path.exists(args.test):
        from sklearn.model_selection import train_test_split
        print("[WARN] test file not found — splitting train 80/20")
        parts_te = []
        for n in sorted(dw_tr['num_nodes'].unique()):
            grp = dw_tr[dw_tr['num_nodes'] == n]
            _, te = train_test_split(grp, test_size=0.2,
                                     random_state=args.seed,
                                     stratify=grp['oc_label'])
            parts_te.append(te)
        dw_te = ensure_cols(pd.concat(parts_te).reset_index(drop=True))
    else:
        dw_te = ensure_cols(pd.read_csv(args.test))

    dl_te = wide_to_long(dw_te, md, ms, msnr, mc)
    print(f"[Test] {len(dw_te)} scenarios -> {len(dl_te)} AUV rows")

    print(f"\n{'='*60}\nSTEP 3: Train (5-fold CV on n=25, refit on all)\n{'='*60}")
    models  = build_models()
    trained = {}
    results = []
    for name, model in models.items():
        m, acc, prec, rec, f1, t = evaluate_model(
            name, model, X_cv, y_cv, X_all, y_all)
        trained[name] = m
        results.append({'Model': name, 'Accuracy': acc, 'Precision': prec,
                        'Recall': rec, 'F1': f1, 'TrainTime_sec': round(t, 2)})

    results_df = pd.DataFrame(results)
    results_df['Rank'] = results_df['F1'].rank(
        ascending=False, method='dense').astype(int)
    results_df = results_df.sort_values('Rank').reset_index(drop=True)
    results_df.to_csv(os.path.join(args.outdir, 'model_results.csv'), index=False)
    print(f"\nOrder: {results_df['Model'].tolist()}")
    print(results_df[['Model','Accuracy','Precision',
                       'Recall','F1']].round(4).to_string(index=False))

    print(f"\n{'='*60}\nSTEP 4: Test metrics and plots\n{'='*60}")
    nc_avail = sorted([n for n in nc
                       if not dw_te[dw_te['num_nodes'] == n].empty])
    cached = {}
    for name, model in trained.items():
        cached[name] = get_metrics(model, dl_te, dw_te, nc_avail)
        print(f"  {name}:"
              f"  PDR={[round(v,1) for v in cached[name][0]]}"
              f"  E2ED={[round(v,1) for v in cached[name][1]]}"
              f"  OH={[round(v,4) for v in cached[name][2]]}"
              f"  Top2%={[round(v,1) for v in cached[name][3]]}")

    base_metrics = get_baseline(dw_te, nc_avail)
    print(f"  No-SDN baseline:"
          f"  PDR={[round(v,1) for v in base_metrics[0]]}"
          f"  E2ED={[round(v,1) for v in base_metrics[1]]}"
          f"  OH={[round(v,4) for v in base_metrics[2]]}")

    plot_performance(cached, base_metrics, nc_avail, args.outdir)
    plot_accuracy(results_df, args.outdir)
    plot_top2_accuracy(cached, nc_avail, args.outdir)
    print_summary(results_df, cached, base_metrics, nc_avail)

    print(f"\nDone -> {args.outdir}/")
    print("  performance_vs_nodes.png   — PDR / E2ED / Overhead (3 panels)")
    print("  model_accuracy.png         — SVM/DTC/RF classification bars")
    print("  top2_accuracy_vs_nodes.png — top-2 OC accuracy vs nodes")
    print("  model_results.csv          — accuracy table")


if __name__ == '__main__':
    main()
